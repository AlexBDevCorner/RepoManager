## Main ViewModel

```csharp
public partial class MainWindowViewModel
    : ObservableObject
{
    private readonly IRepositoryDashboardService _dashboard;

    public ObservableCollection<RepositoryRowViewModel>
        Repositories { get; } = [];

    [ObservableProperty]
    private bool _isBusy;

    public MainWindowViewModel(
        IRepositoryDashboardService dashboard)
    {
        _dashboard = dashboard;
    }

    [RelayCommand]
    private async Task LoadAsync(
        CancellationToken cancellationToken)
    {
        var repositories =
            await _dashboard.LoadAsync(cancellationToken);

        Repositories.Clear();

        foreach (var repository in repositories)
        {
            Repositories.Add(
                new RepositoryRowViewModel(repository));
        }
    }
}
```

Prefer `AsyncRelayCommand` generated by CommunityToolkit. Do not use `async void` except framework event handlers such as application startup.
